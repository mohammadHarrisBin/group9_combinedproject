Stage B — In-Memory Naive SMA (main_stageB.c)
Definition:
Pre-load 1,000,000 trades into a static global array of 
Trade structs in RAM before starting the timer, then process 
indicators using naive O(N) SMA re-summing.

Analysis:
1. Moving CSV parsing out of the timed loop drops 
runtime from 28,709.21 ms (Stage A+) down 
to 277.83 ms (~103x speedup), 
proving disk I/O was the primary bottleneck.

2. Uses a static global array in BSS memory 
and calls sma_calculate_naive(), which loops 
through all 15 and 100 buffer elements on every trade.

3. Matches Stage A with 100% signal parity 
(2292 BUY / 2286 SELL) at 3.60M trades/sec.

Algorithm Design:
BEGIN
    open DEV dataset
    trade_count <- 0

    // Untimed ingestion into RAM
    WHILE trade_count < 1000000 AND read_trade(ts, price, qty, is_buy)
        trades[trade_count] <- (ts, price, qty, is_buy)
        trade_count <- trade_count + 1
    ENDWHILE
    close trades

    vwap <- 0
    ohlc <- 0
    short_sma <- 0
    long_sma <- 0
    sig_st <- 0
    buy_signals <- 0
    sell_signals <- 0
    open "plots/signals_stageB_naive.dat"

    start_time <- now_ns()

    FOR i <- 0 TO trade_count - 1
        READ ts, price, qty from trades[i]
        update vwap, ohlc, short_sma, long_sma
        s_val <- 0
        l_val <- 0
        IF short_sma is full THEN
            s_val <- sma_calculate_naive(short_sma)
        ENDIF
        IF long_sma is full THEN
            l_val <- sma_calculate_naive(long_sma)
        ENDIF
        signal <- SIGNAL_NONE
        IF short_sma is full AND long_sma is full THEN
            signal <- signal_update(sig_st, s_val, l_val)
        ENDIF
        IF signal = SIGNAL_BUY THEN
            buy_signals <- buy_signals + 1
        ENDIF
        IF signal = SIGNAL_SELL THEN
            sell_signals <- sell_signals + 1
        ENDIF
        IF (i + 1) >= 10000 AND (i + 1) <= 12000 THEN
            PRINT (i + 1), price, s_val, l_val, signal to file
        ENDIF
    ENDFOR

    end_time <- now_ns()
    close file
    elapsed_ns <- end_time - start_time
    elapsed_ms <- elapsed_ns / 1000000
    ns_per_trade <- elapsed_ns / trade_count
    trades_per_sec <- trade_count / (elapsed_ns / 1000000000)
    PRINT trade_count, buy_signals, sell_signals, elapsed_ns, elapsed_ms, ns_per_trade, trades_per_sec
END
